-	script	ARPG_Chest3_Drop	-1,{
	end;

OnInit:
	.id = 55002;

	// Primary rate (Lvl 61 - 99, default 3%)
	.@r1$ = replacestr("" + callfunc("F_ModSetting", "arpg-equipments", "chest_3_rate", "3"), ",", ".");
	.@dot1 = strpos(.@r1$, ".");
	.default_rate_low = (.@dot1 == -1 ? (atoi(.@r1$) * 100) : ((atoi(substr(.@r1$, 0, .@dot1 - 1)) * 100) + (atoi(charat(.@r1$, .@dot1 + 1)) * 10)));

	// Secondary rate (Lvl 100 - 150, default 2%)
	.@r2$ = replacestr("" + callfunc("F_ModSetting", "arpg-equipments", "chest_3_secondary_rate", "2"), ",", ".");
	.@dot2 = strpos(.@r2$, ".");
	.rate_secondary = (.@dot2 == -1 ? (atoi(.@r2$) * 100) : ((atoi(substr(.@r2$, 0, .@dot2 - 1)) * 100) + (atoi(charat(.@r2$, .@dot2 + 1)) * 10)));
	end;

OnNPCKillEvent:
	.@lv = getmonsterinfo(killedrid, MOB_LV);
	if (.@lv < 61 || .@lv > 150) end;

	.@main_rate = ($Chest3_Rate > 0) ? $Chest3_Rate : .default_rate_low;

	.@chance = (.@lv <= 99) ? .@main_rate : .rate_secondary;

	if (.@chance > 0 && rand(10000) < .@chance) {
		if (checkweight(.id, 1)) {
			getitem .id, 1;
			dispbottom "You found a Tier 3 Equipment Chest!", 0x3399FF;
		}
	}
	end;
}
