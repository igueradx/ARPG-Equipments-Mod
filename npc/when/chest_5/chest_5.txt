-	script	ARPG_Chest5_Drop	-1,{
	end;

OnInit:
	.id = 55004;

	// Primary rate (Lvl 151 - 200, default 4%)
	.@r1$ = replacestr("" + callfunc("F_ModSetting", "arpg-equipments", "chest_5_rate", "4"), ",", ".");
	.@dot1 = strpos(.@r1$, ".");
	.default_rate_low = (.@dot1 == -1 ? (atoi(.@r1$) * 100) : ((atoi(substr(.@r1$, 0, .@dot1 - 1)) * 100) + (atoi(charat(.@r1$, .@dot1 + 1)) * 10)));

	// Secondary rate (Lvl 201 - 250, default 2%)
	.@r2$ = replacestr("" + callfunc("F_ModSetting", "arpg-equipments", "chest_5_secondary_rate", "2"), ",", ".");
	.@dot2 = strpos(.@r2$, ".");
	.rate_secondary = (.@dot2 == -1 ? (atoi(.@r2$) * 100) : ((atoi(substr(.@r2$, 0, .@dot2 - 1)) * 100) + (atoi(charat(.@r2$, .@dot2 + 1)) * 10)));
	end;

OnNPCKillEvent:
	.@lv = getmonsterinfo(killedrid, MOB_LV);
	if (.@lv < 151 || .@lv > 250) end;

	.@main_rate = ($Chest5_Rate > 0) ? $Chest5_Rate : .default_rate_low;

	.@chance = (.@lv <= 200) ? .@main_rate : .rate_secondary;

	if (.@chance > 0 && rand(10000) < .@chance) {
		if (checkweight(.id, 1)) {
			getitem .id, 1;
			dispbottom "You found a Tier 5 Equipment Chest!", 0xFF9900;
		}
	}
	end;
}
