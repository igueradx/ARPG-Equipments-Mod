-	script	ARPG_Chest6_Drop	-1,{
	end;

OnInit:
	.id = 55005;

	// Single rate (Lvl 200 - 250, default 4%)
	.@r1$ = replacestr("" + callfunc("F_ModSetting", "arpg-equipments", "chest_6_rate", "4"), ",", ".");
	.@dot1 = strpos(.@r1$, ".");
	.default_rate = (.@dot1 == -1 ? (atoi(.@r1$) * 100) : ((atoi(substr(.@r1$, 0, .@dot1 - 1)) * 100) + (atoi(charat(.@r1$, .@dot1 + 1)) * 10)));
	end;

OnNPCKillEvent:
	.@lv = getmonsterinfo(killedrid, MOB_LV);
	if (.@lv < 200 || .@lv > 250) end;

	.@main_rate = ($Chest6_Rate > 0) ? $Chest6_Rate : .default_rate;

	if (.@main_rate > 0 && rand(10000) < .@main_rate) {
		if (checkweight(.id, 1)) {
			getitem .id, 1;
			dispbottom "You found a Tier 6 Legendary Equipment Chest!", 0xFF3333;
		}
	}
	end;
}
