-	script	ARPG_Chest4_Drop	-1,{
	end;

OnInit:
	.id = 55003;

	// Primary rate (Lvl 100 - 150, default 5%)
	.@r1$ = replacestr("" + callfunc("F_ModSetting", "arpg-equipments", "chest_4_rate", "5"), ",", ".");
	.@dot1 = strpos(.@r1$, ".");
	.default_rate_low = (.@dot1 == -1 ? (atoi(.@r1$) * 100) : ((atoi(substr(.@r1$, 0, .@dot1 - 1)) * 100) + (atoi(charat(.@r1$, .@dot1 + 1)) * 10)));

	// Secondary rate (Lvl 151 - 250, default 2%)
	.@r2$ = replacestr("" + callfunc("F_ModSetting", "arpg-equipments", "chest_4_secondary_rate", "2"), ",", ".");
	.@dot2 = strpos(.@r2$, ".");
	.rate_secondary = (.@dot2 == -1 ? (atoi(.@r2$) * 100) : ((atoi(substr(.@r2$, 0, .@dot2 - 1)) * 100) + (atoi(charat(.@r2$, .@dot2 + 1)) * 10)));
	end;

OnNPCKillEvent:
	.@lv = getmonsterinfo(killedrid, MOB_LV);
	if (.@lv < 100 || .@lv > 250) end;

	.@main_rate = ($Chest4_Rate > 0) ? $Chest4_Rate : .default_rate_low;

	.@chance = (.@lv <= 150) ? .@main_rate : .rate_secondary;

	if (.@chance > 0 && rand(10000) < .@chance) {
		if (checkweight(.id, 1)) {
			getitem .id, 1;
			dispbottom "You found a Tier 4 Equipment Chest!", 0xAA66FF;
		}
	}
	end;
}
